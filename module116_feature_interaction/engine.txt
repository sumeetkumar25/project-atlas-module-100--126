# atlas/modules/module116_feature_interaction/engine.py
"""
Module 116 - Feature Interaction Analysis Engine

Measures synergy between feature pairs.
"""

import numpy as np
import pandas as pd
from dataclasses import dataclass
from typing import Optional, Dict, Any, List, Tuple
from itertools import combinations
from sklearn.metrics import roc_auc_score
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import cross_val_score, StratifiedKFold
import warnings
warnings.filterwarnings('ignore')


@dataclass
class FeatureInteractionResult:
    """Results from feature interaction analysis."""
    interaction_table: pd.DataFrame
    interaction_matrix: pd.DataFrame
    top_interactions: pd.DataFrame
    interaction_index: float
    summary_statistics: Dict[str, Any]


class FeatureInteractionEngine:
    """Core engine for feature interaction analysis."""

    IGNORE_COLS = {
        "timestamp", "date", "time", "index",
        "open", "high", "low", "close", "volume",
        "future_breakout"
    }

    SYNERGY_THRESHOLD = 0.25
    MODERATE_THRESHOLD = 0.10
    CONFLICT_THRESHOLD = -0.05

    def __init__(
        self,
        min_samples: int = 100,
        cv_folds: int = 5,
        random_state: int = 42,
        metric: str = "roc_auc"
    ):
        self.min_samples = min_samples
        self.cv_folds = cv_folds
        self.random_state = random_state
        self.metric = metric

    def get_feature_columns(self, df: pd.DataFrame) -> List[str]:
        features = [
            c for c in df.columns
            if c not in self.IGNORE_COLS
            and pd.api.types.is_numeric_dtype(df[c])
            and df[c].nunique() > 1
        ]
        return features

    def generate_pairs(self, features: List[str]) -> List[Tuple[str, str]]:
        return list(combinations(features, 2))

    def compute_feature_score(
        self,
        df: pd.DataFrame,
        feature: str,
        y: np.ndarray
    ) -> float:
        X = pd.to_numeric(
            df[feature],
            errors="coerce"
        ).to_numpy(dtype=float)

        y = pd.to_numeric(
            pd.Series(y, index=df.index),
            errors="coerce"
        ).to_numpy(dtype=float)

        if len(X) != len(y):
            raise ValueError(
                "Feature and target lengths do not match"
            )

        valid_mask = (
            np.isfinite(X)
            & np.isfinite(y)
            & np.isin(y, [0.0, 1.0])
        )

        X = X[valid_mask].reshape(-1, 1)
        y = y[valid_mask].astype(np.int8)

        if len(y) < self.min_samples:
            return 0.0

        if np.unique(X).size < 2:
            return 0.0

        if np.unique(y).size != 2:
            return 0.0

        class_counts = np.bincount(y, minlength=2)

        if np.min(class_counts) < 2:
            return 0.0

        n_splits = min(self.cv_folds, int(np.min(class_counts)))

        if n_splits < 2:
            return 0.0

        try:
            rf = RandomForestClassifier(
                n_estimators=50,
                random_state=self.random_state,
                n_jobs=1
            )

            cv = StratifiedKFold(
                n_splits=n_splits,
                shuffle=True,
                random_state=self.random_state
            )

            if self.metric == 'roc_auc':
                scores = cross_val_score(rf, X, y, cv=cv, scoring='roc_auc')
            elif self.metric == 'accuracy':
                scores = cross_val_score(rf, X, y, cv=cv, scoring='accuracy')
            elif self.metric == 'f1':
                scores = cross_val_score(rf, X, y, cv=cv, scoring='f1')
            else:
                scores = cross_val_score(rf, X, y, cv=cv, scoring='roc_auc')

            return float(np.mean(scores))
        except:
            return 0.0

    def compute_combined_score(
        self,
        df: pd.DataFrame,
        feature_a: str,
        feature_b: str,
        y: np.ndarray
    ) -> float:
        Xa = pd.to_numeric(
            df[feature_a],
            errors="coerce"
        ).to_numpy(dtype=float)

        Xb = pd.to_numeric(
            df[feature_b],
            errors="coerce"
        ).to_numpy(dtype=float)

        y = pd.to_numeric(
            pd.Series(y, index=df.index),
            errors="coerce"
        ).to_numpy(dtype=float)

        if not (len(Xa) == len(Xb) == len(y)):
            raise ValueError(
                "Feature and target lengths do not match"
            )

        valid_mask = (
            np.isfinite(Xa)
            & np.isfinite(Xb)
            & np.isfinite(y)
            & np.isin(y, [0.0, 1.0])
        )

        X = np.column_stack([
            Xa[valid_mask],
            Xb[valid_mask],
        ])
        y = y[valid_mask].astype(np.int8)

        if len(y) < self.min_samples:
            return 0.0

        if np.unique(y).size != 2:
            return 0.0

        class_counts = np.bincount(y, minlength=2)

        if np.min(class_counts) < 2:
            return 0.0

        n_splits = min(self.cv_folds, int(np.min(class_counts)))

        if n_splits < 2:
            return 0.0

        try:
            rf = RandomForestClassifier(
                n_estimators=50,
                random_state=self.random_state,
                n_jobs=1
            )

            cv = StratifiedKFold(
                n_splits=n_splits,
                shuffle=True,
                random_state=self.random_state
            )

            if self.metric == 'roc_auc':
                scores = cross_val_score(rf, X, y, cv=cv, scoring='roc_auc')
            elif self.metric == 'accuracy':
                scores = cross_val_score(rf, X, y, cv=cv, scoring='accuracy')
            elif self.metric == 'f1':
                scores = cross_val_score(rf, X, y, cv=cv, scoring='f1')
            else:
                scores = cross_val_score(rf, X, y, cv=cv, scoring='roc_auc')

            return float(np.mean(scores))
        except:
            return 0.0

    def classify_interaction(self, synergy: float, gain: float) -> str:
        if synergy > self.SYNERGY_THRESHOLD:
            return "Strong Synergy"
        elif synergy > self.MODERATE_THRESHOLD:
            return "Moderate Synergy"
        elif gain < self.CONFLICT_THRESHOLD:
            return "Conflict"
        else:
            return "Independent"

    def compute_interactions(
        self,
        df: pd.DataFrame,
        features: List[str],
        y: np.ndarray
    ) -> pd.DataFrame:
        pairs = self.generate_pairs(features)
        rows = []
        
        print(f"  Computing individual feature scores for {len(features)} features...")
        feature_scores = {
            feature: self.compute_feature_score(df, feature, y)
            for feature in features
        }
        
        print(f"  Computing combined scores for {len(pairs)} pairs...")
        for feature_a, feature_b in pairs:
            score_a = feature_scores[feature_a]
            score_b = feature_scores[feature_b]
            
            combined_score = self.compute_combined_score(
                df, feature_a, feature_b, y
            )
            
            max_individual = max(score_a, score_b)
            gain = combined_score - max_individual
            
            synergy = np.clip(
                gain / (max_individual + 1e-12),
                0.0, 1.0
            )
            
            interaction_type = self.classify_interaction(synergy, gain)
            
            rows.append({
                "feature_a": feature_a,
                "feature_b": feature_b,
                "score_a": score_a,
                "score_b": score_b,
                "combined_score": combined_score,
                "max_individual": max_individual,
                "interaction_gain": gain,
                "synergy_score": synergy,
                "interaction_type": interaction_type
            })
        
        return pd.DataFrame(rows)

    def build_interaction_matrix(
        self,
        interaction_table: pd.DataFrame,
        features: List[str]
    ) -> pd.DataFrame:
        n = len(features)
        matrix = np.zeros((n, n))
        
        feature_to_idx = {f: i for i, f in enumerate(features)}
        
        for _, row in interaction_table.iterrows():
            i = feature_to_idx[row['feature_a']]
            j = feature_to_idx[row['feature_b']]
            matrix[i, j] = row['synergy_score']
            matrix[j, i] = row['synergy_score']
        
        np.fill_diagonal(matrix, 1.0)
        
        return pd.DataFrame(matrix, index=features, columns=features)

    def get_top_interactions(
        self,
        interaction_table: pd.DataFrame,
        top_n: int = 10
    ) -> pd.DataFrame:
        return interaction_table.nlargest(top_n, 'synergy_score')

    def compute_interaction_index(
        self,
        interaction_table: pd.DataFrame
    ) -> float:
        if interaction_table.empty:
            return 0.0
        return float(interaction_table['synergy_score'].mean() * 100)

    def compute(self, df: pd.DataFrame) -> FeatureInteractionResult:
        """
        Main computation method.

        Target extraction is delegated to get_target_series() so this
        module shares the canonical contract for future_breakout with
        every other consumer. The subsequent binary filter is preserved
        because interactions require a strictly {0, 1} target — any
        out-of-set label (e.g. a raw -99 that slipped past an older
        producer) is excluded before scoring.
        """
        from atlas.core.data_contracts import (
            normalize_research_frame,
            get_target_series,
        )

        print("\n  Computing Feature Interactions...")

        # Normalize at the engine boundary.
        df = normalize_research_frame(
            df,
            dataset_name="Module 116",
        )

        if 'future_breakout' not in df.columns:
            raise ValueError("DataFrame must contain 'future_breakout' column")

        # Canonical target extraction.
        target = get_target_series(
            df,
            "future_breakout",
            dataset_name="Module 116",
        )

        valid_target = target.isin([0.0, 1.0])

        df = df.loc[valid_target].copy()
        y = target.loc[valid_target].to_numpy(dtype=np.int8)

        if len(y) < self.min_samples:
            raise ValueError(
                f"Module116 requires at least {self.min_samples} "
                f"valid target rows; received {len(y)}"
            )

        class_counts = np.bincount(y, minlength=2)

        if np.min(class_counts) < 2:
            raise ValueError(
                "Module116 requires at least two observations "
                "in each target class for stratified cross-validation"
            )
        
        features = self.get_feature_columns(df)
        print(f"  Found {len(features)} features")
        
        if len(features) < 2:
            raise ValueError("Need at least 2 features for interaction analysis")
        
        n_pairs = len(features) * (len(features) - 1) // 2
        print(f"  Testing {n_pairs} feature pairs...")
        
        interaction_table = self.compute_interactions(df, features, y)
        
        interaction_matrix = self.build_interaction_matrix(
            interaction_table, features
        )
        
        top_interactions = self.get_top_interactions(interaction_table, top_n=10)
        
        interaction_index = self.compute_interaction_index(interaction_table)
        
        strong_pairs = interaction_table[
            interaction_table['interaction_type'] == 'Strong Synergy'
        ]
        moderate_pairs = interaction_table[
            interaction_table['interaction_type'] == 'Moderate Synergy'
        ]
        conflict_pairs = interaction_table[
            interaction_table['interaction_type'] == 'Conflict'
        ]
        independent_pairs = interaction_table[
            interaction_table['interaction_type'] == 'Independent'
        ]
        
        summary = {
            'total_features': len(features),
            'pairs_tested': len(interaction_table),
            'interaction_index': interaction_index,
            'strong_synergy_pairs': len(strong_pairs),
            'moderate_pairs': len(moderate_pairs),
            'conflicting_pairs': len(conflict_pairs),
            'independent_pairs': len(independent_pairs),
            'best_pair': top_interactions.iloc[0]['feature_a'] + ' + ' + top_interactions.iloc[0]['feature_b'] if not top_interactions.empty else None,
            'best_gain': float(top_interactions.iloc[0]['interaction_gain']) if not top_interactions.empty else 0.0,
            'best_synergy': float(top_interactions.iloc[0]['synergy_score']) if not top_interactions.empty else 0.0
        }
        
        print(f"  Interaction Index: {interaction_index:.2f}/100")
        print(f"  Strong Synergy: {len(strong_pairs)} pairs")
        print(f"  Moderate: {len(moderate_pairs)} pairs")
        print(f"  Conflicts: {len(conflict_pairs)} pairs")
        
        return FeatureInteractionResult(
            interaction_table=interaction_table,
            interaction_matrix=interaction_matrix,
            top_interactions=top_interactions,
            interaction_index=interaction_index,
            summary_statistics=summary
        )
