def create_test_data(n=500, seed=42):
    """Create synthetic test data with interactions."""
    np.random.seed(seed)
    
    # Create features with known interactions
    x1 = np.random.randn(n)
    x2 = np.random.randn(n)
    x3 = np.random.randn(n)
    x4 = np.random.randn(n)  # Noise
    
    # Target with interactions: x1*x2 and x1*x3
    y = (
        0.5 * x1 * x2 + 
        0.3 * x1 * x3 + 
        0.1 * x2 * x3 +
        0.1 * x1 +
        0.1 * x2 +
        np.random.randn(n) * 0.2
    )
    
    # Discretize for classification
    y_binary = (y > np.median(y)).astype(int)
    
    df = pd.DataFrame({
        'feature1': x1,
        'feature2': x2,
        'feature3': x3,
        'feature4': x4,
        'future_breakout': y_binary
    })
    
    return df


def test_module108_target_adapter_integration():
    """
    Verify Module 108 target attachment through the real
    BaseResearchModule dependency-injection workflow.
    """
    print("\n" + "=" * 60)
    print("TEST: Module 108 → Module 110 Target Integration")
    print("=" * 60)

    n = 300

    rng = np.random.default_rng(42)

    timestamps = pd.date_range(
        "2026-01-01",
        periods=n,
        freq="15min"
    )

    market_df = pd.DataFrame({
        "timestamp": timestamps,
        "open": rng.uniform(100, 110, n),
        "high": rng.uniform(110, 120, n),
        "low": rng.uniform(90, 100, n),
        "close": rng.uniform(100, 110, n),
        "volume": rng.uniform(1, 100, n),
        "feature1": rng.normal(size=n),
        "feature2": rng.normal(size=n),
        "feature3": rng.normal(size=n),
    })

    breakout_targets = pd.DataFrame({
        "timestamp": timestamps,
        "breakout_5": rng.choice([-1, 0, 1, -99], size=n),
        "breakout_10": rng.choice([-1, 0, 1, -99], size=n),
    })

    research = FeatureInteractionResearch(
        target_column="future_breakout",
        target_horizon=5,
        n_bootstrap=2,
        n_folds=2,
    )

    # ----------------------------------------------------------
    # Exercise the real BaseResearchModule dependency workflow:
    #
    #   1. load_dataframe()         → populates market_df, calls
    #                                 normalize_research_frame()
    #   2. set_dependency_features() → validates dependency
    #                                 timestamps, saves them to
    #                                 _dependency_timestamps, drops
    #                                 the timestamp column from
    #                                 feature_df, merges into df
    #   3. _prepare_data()          → reconstructs breakout_targets
    #                                 from feature_df + _dependency_ts,
    #                                 calls attach_breakout_target()
    # ----------------------------------------------------------
    research.load_dataframe(
        market_df,
        dataset_id="module110_adapter_test",
    )

    research.set_dependency_features(breakout_targets)

    prepared = research._prepare_data()
    prepared_df = prepared["df"]

    # ----------------------------------------------------------
    # Assert: future_breakout is attached.
    # ----------------------------------------------------------
    assert "future_breakout" in prepared_df.columns, (
        "Module 108 target was not attached as future_breakout"
    )

    # ----------------------------------------------------------
    # Assert: attached values match breakout_5 exactly.
    #
    # The adapter copies breakout_{horizon} values verbatim
    # (including -99 labels). Filtering of invalid labels happens
    # later in engine.prepare_data(), not in the adapter.
    # ----------------------------------------------------------
    assert np.array_equal(
        prepared_df["future_breakout"].to_numpy(),
        breakout_targets["breakout_5"].to_numpy(),
    ), (
        "future_breakout values do not match breakout_5"
    )

    # ----------------------------------------------------------
    # Assert: breakout_* columns are excluded from predictors.
    #
    # detect_features() must exclude future-derived Module 108
    # targets to prevent target leakage into interaction analysis.
    # ----------------------------------------------------------
    detected = research.engine.detect_features(prepared_df)

    assert "breakout_5" not in detected, (
        "breakout_5 leaked into Module 110 predictors"
    )
    assert "breakout_10" not in detected, (
        "breakout_10 leaked into Module 110 predictors"
    )

    print("  ✓ Module 108 breakout_5 correctly attached as future_breakout")
    print("  ✓ breakout_5 and breakout_10 excluded from predictors")

    return True


def test_feature_names_with_underscores():
    """
    Regression test ensuring interaction pairs remain unambiguous
    when feature names contain underscores.
    """

    print("\n" + "=" * 60)
    print("TEST: Feature Names Containing Underscores")
    print("=" * 60)

    n = 200

    rng = np.random.default_rng(42)

    df = pd.DataFrame({
        "return_15m": rng.normal(size=n),
        "volatility_1h": rng.normal(size=n),
        "rsi_14": rng.normal(size=n),
        "future_breakout": rng.choice(
            [-1, 0, 1],
            size=n
        ),
    })

    engine = FeatureInteractionEngine(
        n_folds=2,
        n_bootstrap=1,
    )

    result = engine.compute(df)

    assert not result.interaction_scores.empty

    for pair in result.interaction_scores.index:
        parts = pair.split("||")

        assert len(parts) == 2

    assert (
        "return_15m||volatility_1h"
        in result.interaction_scores.index
    )

    print(
        "  ✓ Underscore-containing feature names "
        "remain correctly represented"
    )

    return True


def test_breakout_target_is_classification():
    """
    Verify Module 110 accepts the Module 108 target convention:
        -1 = DOWN
         0 = NO BREAKOUT
        +1 = UP
    """

    print("\n" + "=" * 60)
    print("TEST: Breakout Classification Target")
    print("=" * 60)

    n = 300

    rng = np.random.default_rng(42)

    df = pd.DataFrame({
        "feature_a": rng.normal(size=n),
        "feature_b": rng.normal(size=n),
        "future_breakout": rng.choice(
            [-1, 0, 1],
            size=n
        ),
    })

    engine = FeatureInteractionEngine(
        n_folds=2,
        n_bootstrap=1,
    )

    X, y, features = engine.prepare_data(df)

    assert set(np.unique(y)).issubset(
        {-1, 0, 1}
    )

    result = engine.compute(df)

    assert not result.interaction_scores.empty

    print(
        "  ✓ -1/0/+1 breakout classification "
        "processed successfully"
    )

    return True


def test_future_breakout_columns_excluded():
    """Test that Module 108 breakout_* columns cannot become features."""
    print("\n" + "=" * 60)
    print("TEST: Future Breakout Columns Excluded")
    print("=" * 60)

    df = create_test_data()

    # Simulate Module 108 future-derived targets.
    df["breakout_5"] = np.random.randint(-1, 2, len(df))
    df["breakout_10"] = np.random.randint(-1, 2, len(df))
    df["breakout_20"] = np.random.randint(-1, 2, len(df))

    engine = FeatureInteractionEngine(
        target_column="future_breakout"
    )

    features = engine.detect_features(df)

    assert "breakout_5" not in features
    assert "breakout_10" not in features
    assert "breakout_20" not in features

    assert "feature1" in features
    assert "feature2" in features
    assert "feature3" in features

    print(f"  Detected features: {features}")
    print("  ✓ Future breakout exclusion test passed")

    return True


def test_future_breakout_explicit_feature_rejected():
    """Test that explicitly supplied breakout_* features are rejected."""
    print("\n" + "=" * 60)
    print("TEST: Explicit Future Breakout Leakage Rejected")
    print("=" * 60)

    df = create_test_data()

    df["breakout_5"] = np.random.randint(-1, 2, len(df))

    engine = FeatureInteractionEngine(
        target_column="future_breakout",
        feature_columns=[
            "feature1",
            "feature2",
            "breakout_5",
        ]
    )

    try:
        engine.prepare_data(df)

    except ValueError as e:
        message = str(e)

        assert "Future-derived Module 108 target columns" in message
        assert "breakout_5" in message

        print("  ✓ Explicit future breakout leakage rejected")
        return True

    raise AssertionError(
        "Module 110 accepted breakout_5 as a predictor"
    )

def test_engine_normalization():
    """Test 2: Score normalization."""
    print("\n" + "=" * 60)
    print("TEST 2: Score Normalization")
    print("=" * 60)
    
    engine = FeatureInteractionEngine()
    
    scores = pd.Series([10, 20, 30, 40, 50])
    normalized = engine.normalize_scores(scores)
    
    assert np.isclose(normalized.min(), 0)
    assert np.isclose(normalized.max(), 1)
    assert np.all((0 <= normalized) & (normalized <= 1))
    
    scores = pd.Series([5, 5, 5, 5, 5])
    normalized = engine.normalize_scores(scores)
    assert np.all(normalized == 0)
    
    print("  ✓ Normalization test passed")
    return True

def test_engine_ranking():
    """Test 7: Ranking and combination."""
    print("\n" + "=" * 60)
    print("TEST 7: Ranking and Combination")
    print("=" * 60)
    
    engine = FeatureInteractionEngine()
    
    scores_dict = {
        'method1': pd.Series([0.8, 0.6, 0.4, 0.2], index=['A||B', 'A||C', 'B||C', 'C||D']),
        'method2': pd.Series([0.7, 0.5, 0.3, 0.1], index=['A||B', 'A||C', 'B||C', 'C||D']),
        'method3': pd.Series([0.9, 0.4, 0.6, 0.2], index=['A||B', 'A||C', 'B||C', 'C||D'])
    }
    
    combined = engine.combine_scores(scores_dict)
    
    assert 'interaction_score' in combined.columns
    assert len(combined) == 4
    assert not combined.isna().any().any()
    assert np.all((0 <= combined['interaction_score']) & (combined['interaction_score'] <= 1))
    assert combined.index[0] == 'A||B'
    
    print("  ✓ Ranking test passed")
    return True

            result,
            tmp_path / "detailed.json"

    print("  ✓ Exporter test passed")
