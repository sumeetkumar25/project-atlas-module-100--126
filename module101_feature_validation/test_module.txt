import sys

sys.path.insert(0, str(Path(__file__).parent.parent.parent.parent))

import numpy as np
import pandas as pd

from atlas.modules.module101_feature_validation.engine import FeatureValidationEngine
from atlas.modules.module101_feature_validation.statistics import FeatureValidationStatisticsEngine

from atlas.modules.module101_feature_validation.report import FeatureValidationReport
from atlas.modules.module101_feature_validation.module import FeatureValidationResearch


def test_random_feature():
    """Test 1: Random feature → IC ≈ 0."""
    print("\n" + "=" * 60)
    print("TEST 1: Random Feature (No Predictive Power)")
    print("=" * 60)

    engine = FeatureValidationEngine(horizon=10, buckets=10)
    stats_engine = FeatureValidationStatisticsEngine()

    np.random.seed(42)
    n = 5000
    close = 100 + np.cumsum(np.random.normal(0, 1, n))
    feature = np.random.normal(0, 1, n)

    result = engine.compute(close, feature, series_type="random")
    stats = stats_engine.analyze(result)

    assert abs(result.information_coefficient) < 0.05, (
        f"Random feature IC should be near 0, got {result.information_coefficient:.6f}"
    )

    print(f"  IC: {result.information_coefficient:.6f}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Random feature test passed")
    return True


def test_perfect_predictor():
    """Test 2: Perfect predictor → IC ≈ 1."""
    print("\n" + "=" * 60)
    print("TEST 2: Perfect Predictor")
    print("=" * 60)

    engine = FeatureValidationEngine(horizon=10, buckets=10)
    stats_engine = FeatureValidationStatisticsEngine()

    np.random.seed(42)
    n = 5000
    close = 100 + np.cumsum(np.random.normal(0, 1, n))
    future_returns = close[10:] / close[:-10] - 1.0
    feature = np.concatenate([future_returns, np.zeros(10)])

    result = engine.compute(close, feature, series_type="perfect")
    stats = stats_engine.analyze(result)

    assert result.information_coefficient > 0.90, (
        f"Perfect predictor IC should be near 1, got {result.information_coefficient:.6f}"
    )

    print(f"  IC: {result.information_coefficient:.6f}")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Perfect predictor test passed")
    return True

def test_pipeline():
    """Test 6: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 6: Full Pipeline")
    print("=" * 60)

    n = 500
    np.random.seed(42)
    df = pd.DataFrame({
        "close": 100 + np.cumsum(np.random.normal(0, 1, n)),
        "feature": np.random.normal(0, 1, n),
        "timestamp": pd.date_range("2020-01-01", periods=n, freq="15min"),
        "open": np.zeros(n),
        "high": np.zeros(n),
        "low": np.zeros(n),
        "volume": np.ones(n),
    })

    # Module 101 requires the Module 100 `trend_score` dependency column.
    # We synthesize it here from close prices (rolling mean of returns)
    # so the fixture satisfies the dependency contract without touching
    # production code and without introducing look-ahead.
    df["trend_score"] = (
        df["close"]
        .pct_change()
        .fillna(0.0)
        .rolling(20, min_periods=1)
        .mean()
    )

    research = FeatureValidationResearch(horizon=10)
    result = research.run_dataframe(df, dataset_id="pipeline_test")

    assert result.status == "completed"
    assert "feature_validation_statistics" in result.artifacts
    assert "bucket_statistics" in result.artifacts
    assert "feature_validation_report" in result.artifacts
    assert result.statistics.get("validation_passed") == True

    print(f"  IC: {result.statistics.get('information_coefficient'):.6f}")
    print("  ✓ Full pipeline test passed")
    return True
