def create_test_data(n=1000, seed=42):
    """Create synthetic test data."""
    np.random.seed(seed)

    # Create features with some signal
    trend_score = np.random.uniform(0, 100, n)
    compression_score = np.random.uniform(0, 100, n)
    hurst_exponent_rolling_60d = np.random.uniform(
        0.35, 0.75, n
    )
    normalized_entropy_rolling_60d = np.random.uniform(
        0.40, 0.95, n
    )
    dfa_alpha = np.random.uniform(0.2, 0.8, n)
    mutual_information = np.random.uniform(0, 1, n)

    # Create close price with some trend
    close = 100 + np.cumsum(np.random.randn(n) * 0.5)

    df = pd.DataFrame({
        "trend_score": trend_score,
        "compression_score": compression_score,
        "hurst_exponent_rolling_60d": hurst_exponent_rolling_60d,
        "normalized_entropy_rolling_60d": normalized_entropy_rolling_60d,
        "dfa_alpha": dfa_alpha,
        "mutual_information": mutual_information,
        "close": close
    })

    return df


def test_engine_causal_thresholds():
    """Test 1: Expanding thresholds are causal (past-only)."""
    print("\n" + "=" * 60)
    print("TEST 1: Causal Thresholds")
    print("=" * 60)

    engine = FeatureCombinationEngine()

    np.random.seed(42)
    n = 500

    # Monotonic entropy series — allows us to reason about the
    # expanding quantile behavior deterministically.
    entropy = np.linspace(0.0, 1.0, n)
    hurst = np.linspace(0.0, 1.0, n)

    df = pd.DataFrame({
        "trend_score": np.zeros(n),
        "compression_score": np.zeros(n),
        "hurst_exponent_rolling_60d": hurst,
        "normalized_entropy_rolling_60d": entropy,
    })

    entropy_threshold, hurst_threshold = engine._compute_expanding_thresholds(df)

    # Rows before min_periods must be NaN (no history yet).
    assert entropy_threshold.isna().iloc[0], (
        "First threshold row must be NaN — no history"
    )

    # Once enough history exists, the threshold must be defined.
    assert not entropy_threshold.isna().iloc[-1], (
        "Final threshold row must be finite — enough history"
    )

    # Causality check: the threshold at row t must depend only on
    # rows < t. If we perturb row t+k, thresholds at rows <= t must
    # be unchanged.
    perturbed_entropy = entropy.copy()
    perturbed_entropy[400] = 999.0

    df_perturbed = df.copy()
    df_perturbed["normalized_entropy_rolling_60d"] = perturbed_entropy

    entropy_threshold_p, _ = engine._compute_expanding_thresholds(df_perturbed)

    # Thresholds before row 400 must be identical between runs.
    for t in [0, 100, 200, 300, 399]:
        assert np.isclose(
            entropy_threshold.iloc[t] if not np.isnan(entropy_threshold.iloc[t]) else 0.0,
            entropy_threshold_p.iloc[t] if not np.isnan(entropy_threshold_p.iloc[t]) else 0.0,
        ) or (
            np.isnan(entropy_threshold.iloc[t])
            and np.isnan(entropy_threshold_p.iloc[t])
        ), (
            f"Threshold at row {t} changed when row 400 changed — "
            "thresholds are not causal"
        )

    # Thresholds at rows >= 401 may differ (they legitimately see the
    # perturbed row).
    # We don't assert inequality because the Q25 quantile may be robust
    # to a single outlier at the 400th rank; equality here is fine.

    print(f"  Warm-up rows (NaN threshold): "
          f"{entropy_threshold.isna().sum()}")
    print(f"  Valid threshold rows: "
          f"{(~entropy_threshold.isna()).sum()}")
    print("  ✓ Causal thresholds test passed")
    return True


def test_engine_state_creation():
    """Test 2: State creation produces valid states."""
    print("\n" + "=" * 60)
    print("TEST 2: State Creation")
    print("=" * 60)

    engine = FeatureCombinationEngine()

    # The engine now uses causal expanding-quantile thresholds for
    # entropy and Hurst, so a meaningful number of warm-up bars is
    # required before any state can be assigned. 500 rows is enough
    # to produce several hundred classified rows while staying fast.
    n = 500
    np.random.seed(42)

    df = pd.DataFrame({
        "trend_score": np.random.uniform(0, 100, n),
        "compression_score": np.random.uniform(0, 100, n),
        "normalized_entropy_rolling_60d": np.random.uniform(0.40, 0.95, n),
        "hurst_exponent_rolling_60d": np.random.uniform(0.35, 0.75, n),
    })

    entropy_threshold, hurst_threshold = engine._compute_expanding_thresholds(df)
    state = engine.create_state(df, entropy_threshold, hurst_threshold)

    # Array length must match input.
    assert len(state) == n, (
        f"State length {len(state)} != input length {n}"
    )

    # Non-NaN states must be integers in [0, 15].
    finite = state[np.isfinite(state)]
    assert len(finite) > 0, "Should have at least some classified rows"
    assert np.all((finite >= 0) & (finite <= 15)), (
        f"States out of [0, 15]: {np.unique(finite)}"
    )

    # Values must be integer-valued (as floats).
    assert np.allclose(finite, np.round(finite)), (
        "States should be integer-valued"
    )

    # Warm-up rows must be NaN (threshold not yet established).
    assert np.isnan(state[0]), "First row should be NaN (no history)"

    # With 500 random rows we should observe multiple distinct states.
    unique_states = np.unique(finite)
    assert len(unique_states) > 1, (
        f"Expected multiple states, got {unique_states}"
    )

    print(f"  Classified rows: {len(finite)}/{n}")
    print(f"  Unique states: {sorted(int(s) for s in unique_states)}")
    print("  ✓ State creation test passed")
    return True


def test_engine_full():
    """Test 3: Full engine computation."""
    print("\n" + "=" * 60)
    print("TEST 3: Full Engine Computation")
    print("=" * 60)

    df = create_test_data(n=1000)
    engine = FeatureCombinationEngine(future_horizon=5, min_count=20)

    features = df[engine.feature_columns]
    close = df["close"]

    result = engine.compute(features, close)

    # Check results
    assert result.total_states > 0, "Should have some states"
    assert result.sample_size > 0, "Should have samples"
    assert result.best_state >= 0, "Should have a best state"
    assert result.information_coefficient != 0, "IC should be calculated"

    # Check state statistics
    assert not result.state_statistics.empty, "State statistics should not be empty"

    print(f"  Total States: {result.total_states}")
    print(f"  Significant States: {result.significant_states}")
    print(f"  Best State: {result.best_state}")
    print(f"  IC: {result.information_coefficient:.4f}")
    print("  ✓ Full engine test passed")
    return True


def test_statistics():
    """Test 4: Statistics engine."""
    print("\n" + "=" * 60)
    print("TEST 4: Statistics Engine")
    print("=" * 60)

    df = create_test_data(n=1000)
    engine = FeatureCombinationEngine(future_horizon=5, min_count=20)
    stats_engine = FeatureCombinationStatisticsEngine()

    result = engine.compute(
        df[engine.feature_columns],
        df["close"]
    )

    stats = stats_engine.analyze(result)

    # Check statistics
    assert stats.total_states == result.total_states
    assert stats.significant_states == result.significant_states
    assert stats.best_state == result.best_state
    assert stats.predictive_power in ["Strong", "Moderate", "Weak", "None"]

    print(f"  Predictive Power: {stats.predictive_power}")
    print(f"  Total States: {stats.total_states}")
    print("  ✓ Statistics test passed")
    return True
