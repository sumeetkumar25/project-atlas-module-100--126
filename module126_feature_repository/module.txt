# atlas/modules/module126_feature_repository/module.py
"""
Module 126
Feature Repository

Collects every engineered feature into one standardized dataframe.

Pipeline: load feature store → merge features → validate → statistics → validation → report → export
"""

from typing import Optional, Dict, Any
import pandas as pd
from pathlib import Path

from atlas.core.base_research_module import BaseResearchModule
from atlas.framework.registry import AtlasRegistry

from .engine import FeatureRepositoryEngine
from .statistics import FeatureRepositoryStatisticsEngine
from .validator import run_validation
from .report import FeatureRepositoryReport
from .exporter import FeatureRepositoryExporter


class FeatureRepositoryResearch(BaseResearchModule):
    """Main research module for feature repository."""

    MODULE_OUTPUT = "module126"

    METADATA = {
        **FeatureRepositoryStatisticsEngine.METADATA,
        "depends_on": [
            "100",
            "102",
            "035",
            "037",
        ],
    }

    def __init__(
        self,
        registry: Optional[AtlasRegistry] = None,
        timestamp_column: str = "timestamp",
        drop_duplicates: bool = True,
        fill_method: str = "ffill"
    ):
        """
        Initialize the feature repository module.
        
        Args:
            registry: Optional Atlas registry
            timestamp_column: Name of the timestamp column
            drop_duplicates: Whether to drop duplicate timestamps
            fill_method: Method for filling missing values ('ffill', 'bfill', 'none')
        """
        super().__init__(
            module_name="Feature Repository",
            module_version=self.METADATA["version"],
        )
        
        self.registry = registry or AtlasRegistry()
        self.engine = FeatureRepositoryEngine(
            timestamp_column=timestamp_column,
            drop_duplicates=drop_duplicates,
            fill_method=fill_method
        )
        self.stats_engine = FeatureRepositoryStatisticsEngine()
        self.result_data = None
        self.statistics = None
        self.feature_list_report = None
        self._output_dir = None
        self._feature_dict = None
        self._feature_store = None

    def _register_pipeline_steps(self):
        """Register pipeline steps."""
        self.pipeline_steps = [
            ("feature_repository", self._build_repository),
            ("statistics", self._compute_statistics),
            ("validate", self._validate),
            ("export", self._export_results),
        ]
    
    def _prepare_data(self) -> Dict[str, Any]:
        """Prepare data from DataFrame."""
        # For this module, we expect a feature store path or dictionary
        if self._feature_store is not None:
            return {"feature_store": self._feature_store}
        elif self._feature_dict is not None:
            return {"feature_dict": self._feature_dict}
        else:
            raise ValueError("No feature data provided. Use run_repository() with feature_store or feature_dict.")
    
    def _build_repository(self) -> Dict[str, Any]:
        """Step 1: Build feature repository."""
        self.logger.info("Step 1: Building feature repository")
        
        prepared = self._prepare_data()
        
        if "feature_store" in prepared:
            self.result_data = self.engine.compute(
                feature_store=prepared["feature_store"]
            )
        else:
            self.result_data = self.engine.compute(
                feature_dict=prepared["feature_dict"]
            )
        
        self.logger.metric(
            "Total Rows",
            f"{self.result_data.summary_statistics['total_rows']:,}"
        )
        self.logger.metric(
            "Total Features",
            str(self.result_data.summary_statistics['total_features'])
        )
        self.logger.metric(
            "Missing %",
            f"{self.result_data.summary_statistics['missing_percentage']:.2f}%"
        )
        self.logger.metric(
            "Memory",
            f"{self.result_data.summary_statistics['memory_usage_mb']:.2f} MB"
        )
        self.logger.success("Feature repository built")
        
        # Generate report
        self.feature_list_report = FeatureRepositoryReport.generate_feature_list(
            self.result_data.feature_metadata,
            top_n=20
        )
        
        return {
            "total_rows": self.result_data.summary_statistics["total_rows"],
            "total_features": self.result_data.summary_statistics["total_features"],
            "missing_percentage": self.result_data.summary_statistics["missing_percentage"],
            "memory_usage_mb": self.result_data.summary_statistics["memory_usage_mb"]
        }
    
    def _compute_statistics(self) -> Dict[str, Any]:
        """Step 2: Computing statistics."""
        self.logger.info("Step 2: Computing statistics")
        self.statistics = self.stats_engine.analyze(self.result_data)
        
        self.logger.metric(
            "Feature Density",
            f"{self.statistics.feature_density:.2%}"
        )
        self.logger.metric(
            "Duration",
            f"{self.statistics.time_range_duration_days} days"
        )
        self.logger.success("Statistics computed")
        
        return self.stats_engine.summary_dict(self.statistics)
    
    def _validate(self) -> Dict[str, Any]:
        """Step 3: Validation."""
        self.logger.info("Step 3: Validation")
        passed = run_validation(self.result_data, self.statistics)
        self.logger.success("Validation complete")
        return {"software_validation_passed": passed}
    
    def _export_results(self) -> Dict[str, Any]:
        """Step 4: Export."""
        self.logger.info("Step 4: Export")
        
        # Register DataFrames as artifacts
        self.result.artifacts["master_feature_matrix"] = self.result_data.master_feature_matrix
        self.result.artifacts["feature_metadata"] = self.result_data.feature_metadata
        
        # Store summary
        summary_df = pd.DataFrame([self.result_data.summary_statistics])
        self.result.artifacts["repository_summary"] = summary_df
        
        # Generate and register reports
        main_report = FeatureRepositoryReport.generate(
            self.statistics,
            metadata=self.METADATA,
            dataset_name=self.result.dataset_identifier,
        )
        self.result.artifacts["feature_repository_report"] = main_report
        self.result.artifacts["feature_list_report"] = self.feature_list_report
        
        # Export to files
        if self._output_dir is not None:
            export_dir = self._output_dir / "exports"
            export_dir.mkdir(parents=True, exist_ok=True)
            
            FeatureRepositoryExporter.export_master_matrix(
                self.result_data.master_feature_matrix,
                export_dir / "master_feature_matrix.csv"
            )
            
            FeatureRepositoryExporter.export_metadata(
                self.result_data.feature_metadata,
                export_dir / "feature_metadata.csv"
            )
            
            FeatureRepositoryExporter.export_summary(
                self.result_data.summary_statistics,
                export_dir / "repository_summary.csv"
            )
        
        self.logger.metric("Artifacts", str(len(self.result.artifacts)))
        self.logger.success("Export complete")
        
        return {
            "export_status": "complete",
            "artifact_count": len(self.result.artifacts),
            "total_features": self.result_data.summary_statistics["total_features"],
            "total_rows": self.result_data.summary_statistics["total_rows"]
        }
    
    def compute(self) -> Dict[str, Any]:
        """Run the full computation pipeline."""
        self.logger.section("Feature Repository Pipeline")
        results = self.execute_pipeline()
        self.result.status = "completed"
        self.logger.success("Module 126 complete")
        return results
    
    def run(self, filepath: str, output_dir=None):
        """Run module on file."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run(filepath, output_dir)
    
    def run_dataframe(self, df, dataset_id="in_memory", output_dir=None):
        """Run module on a single DataFrame."""
        self._output_dir = Path(output_dir) if output_dir else None
        return super().run_dataframe(df, dataset_id, output_dir)
    
    def run_repository(
        self,
        feature_store: Optional[Path] = None,
        feature_dict: Optional[Dict[str, pd.DataFrame]] = None,
        dataset_id: str = "feature_repository",
        output_dir: Optional[str] = None
    ) -> Any:
        """
        Run the repository on a feature store or dictionary.
        
        Args:
            feature_store: Path to the feature store directory
            feature_dict: Dictionary mapping feature name to DataFrame
            dataset_id: Identifier for the dataset
            output_dir: Output directory for results
        """
        self._feature_store = feature_store
        self._feature_dict = feature_dict
        self._output_dir = Path(output_dir) if output_dir else None
        
        # Create a dummy DataFrame to satisfy BaseResearchModule
        dummy = pd.DataFrame(
            {
                "timestamp": [pd.Timestamp("2000-01-01")]
            }
        )
        
        # Let BaseResearchModule.run_dataframe() initialize everything
        return super().run_dataframe(
            dummy,
            dataset_id=dataset_id,
            output_dir=output_dir,
        )
