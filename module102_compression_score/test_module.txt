def test_highly_compressed():
    """Test 1: Highly compressed market → Strong Compression regime."""
    print("\n" + "=" * 60)
    print("TEST 1: Highly Compressed Market")
    print("=" * 60)

    engine = CompressionScoreEngine(
        atr_weight=0.25, std_weight=0.25, entropy_weight=0.25, range_weight=0.25
    )
    stats_engine = CompressionScoreStatisticsEngine()

    result = engine.compute(
        atr_percentile=0.10,
        std_percentile=0.08,
        normalized_entropy=0.72,
        range_ratio=0.40,
        sample_size=10000,
    )
    stats = stats_engine.analyze(result)

    # With equal weights: (0.90 + 0.92 + 0.28 + 0.60) / 4 * 100 = 67.5
    # This should be in Strong Compression range (60-80)
    assert result.compression_score >= 60, (
        f"Highly compressed should score >= 60, got {result.compression_score:.2f}"
    )
    assert result.compression_score < 80, (
        f"Score should be < 80 for Strong Compression, got {result.compression_score:.2f}"
    )
    assert stats.regime == "Strong Compression", (
        f"Expected 'Strong Compression', got '{stats.regime}'"
    )

    print(f"  Compression Score: {result.compression_score:.2f}/100")
    print(f"  Regime: {stats.regime}")
    print("  ✓ Highly compressed test passed")
    return True

def test_custom_weights():
    """Test 4: Custom weights change the score."""
    print("\n" + "=" * 60)
    print("TEST 4: Custom Weights")
    print("=" * 60)

    engine_default = CompressionScoreEngine(
        atr_weight=0.25, std_weight=0.25, entropy_weight=0.25, range_weight=0.25
    )
    engine_range_heavy = CompressionScoreEngine(
        atr_weight=0.10, std_weight=0.10, entropy_weight=0.10, range_weight=0.70
    )

    result_default = engine_default.compute(
        atr_percentile=0.50, std_percentile=0.50,
        normalized_entropy=0.90, range_ratio=0.30, sample_size=10000,
    )
    result_range_heavy = engine_range_heavy.compute(
        atr_percentile=0.50, std_percentile=0.50,
        normalized_entropy=0.90, range_ratio=0.30, sample_size=10000,
    )

    assert result_range_heavy.compression_score > result_default.compression_score, (
        "Range-heavy weights should produce higher score for low range ratio"
    )

    print(f"  Default weights score:  {result_default.compression_score:.2f}")
    print(f"  Range-heavy score:      {result_range_heavy.compression_score:.2f}")
    print("  ✓ Custom weights test passed")
    return True


def test_validator():
    """Test 5: Validator."""
    print("\n" + "=" * 60)
    print("TEST 5: Validator")
    print("=" * 60)

    engine = CompressionScoreEngine()
    stats_engine = CompressionScoreStatisticsEngine()

    result = engine.compute(
        atr_percentile=0.50, std_percentile=0.50,
        normalized_entropy=0.90, range_ratio=1.0, sample_size=5000,
    )
    stats = stats_engine.analyze(result)

    passed = run_validation(result, stats)
    assert passed
    print("  ✓ Validator test passed")
    return True


def test_pipeline():
    """Test 6: Full pipeline."""
    print("\n" + "=" * 60)
    print("TEST 6: Full Pipeline")
    print("=" * 60)

    df = pd.DataFrame([{
        "timestamp": pd.Timestamp("2024-01-01 00:00:00"),
        "atr_percentile": 0.15,
        "std_percentile": 0.12,
        "normalized_entropy": 0.75,
        "range_ratio": 0.45,
        "sample_size": 10000,
    }])

    research = CompressionScoreResearch()
    result = research.run_dataframe(df, dataset_id="pipeline_test")

    assert result.status == "completed"
    assert "compression_score_statistics" in result.artifacts
    assert "compression_score_report" in result.artifacts
    assert "features" in result.artifacts
    assert result.statistics.get("validation_passed") == True

    feature_df = result.artifacts["features"]
    assert list(feature_df.columns) == [
        "timestamp",
        "compression_score",
    ]
    assert len(feature_df) == len(df)
    assert feature_df["timestamp"].equals(df["timestamp"])
    assert feature_df["compression_score"].notna().all()

    print(f"  Compression Score: {result.statistics.get('compression_score'):.2f}/100")
    print("  ✓ Full pipeline test passed")
    return True
