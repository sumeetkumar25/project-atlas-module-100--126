def create_test_data(n=500, seed=42):
    """Create synthetic test data with a known weak edge."""
    np.random.seed(seed)

    # Features with a real signal
    x1 = np.random.randn(n)
    x2 = np.random.randn(n) * 0.5 + x1 * 0.3
    x3 = np.random.randn(n) * 0.3 + x1 * 0.2
    x4 = np.random.randn(n)  # Pure noise

    # Target with signal + noise
    y_prob = 1 / (1 + np.exp(-(0.5 * x1 + 0.3 * x2 + 0.1 * x3)))
    y = (np.random.random(n) < y_prob).astype(int)

    # Create DataFrame
    df = pd.DataFrame({
        'feature1': x1,
        'feature2': x2,
        'feature3': x3,
        'feature4': x4,
        'future_breakout': y
    })

    return df

def test_bootstrap_cv_duplicate_leakage():
    """
    Bootstrap duplicate-source leakage diagnostic.

    This test verifies whether a bootstrap sample can place the
    same original source row into both the training and validation
    sets of a cross-validation split.

    Bootstrap sampling with replacement is expected to produce
    duplicate source rows inside the bootstrapped dataset.

    If those duplicates then appear on both sides of a
    cross-validation split, the model is trained on rows that are
    effectively identical to rows used for validation. That
    inflates the measured score and produces the pathological
    bootstrap bias we observed.

    This test runs quickly because:
        - sample size = 200
        - bootstrap iterations = 1
        - cv folds = 3
        - trees = 10
    """

    print("\n" + "=" * 60)
    print("TEST: Bootstrap Duplicate-Source Leakage Diagnostic")
    print("=" * 60)

    n = 200

    rng = np.random.default_rng(42)

    # ---------------------------------------------------------
    # Build a small dataset with a stable original-row identity
    # ---------------------------------------------------------
    original_ids = np.arange(n)

    X = pd.DataFrame({
        "feature_1": rng.normal(size=n),
        "feature_2": rng.normal(size=n),
        "feature_3": rng.normal(size=n),
    })

    # A simple target with a weak but real signal
    signal = (
        0.6 * X["feature_1"]
        + 0.4 * X["feature_2"]
    )

    y = (
        signal
        + rng.normal(0, 0.5, size=n)
        > 0
    ).astype(int)

    # ---------------------------------------------------------
    # Bootstrap sample (single iteration)
    # ---------------------------------------------------------
    bootstrap_rng = np.random.default_rng(1234)

    bootstrap_indices = bootstrap_rng.integers(
        0,
        n,
        size=n,
    )

    boot_original_ids = original_ids[bootstrap_indices]

    # Sanity check: bootstrap sampling with replacement
    # should produce duplicate source rows.
    unique_boot_ids = np.unique(boot_original_ids)

    duplicate_count = n - len(unique_boot_ids)

    print(
        f"  Bootstrap sample size: {n}"
    )

    print(
        f"  Unique original rows in bootstrap: "
        f"{len(unique_boot_ids)}"
    )

    print(
        f"  Duplicate source rows produced: "
        f"{duplicate_count}"
    )

    assert duplicate_count > 0, (
        "Bootstrap sampling with replacement should produce "
        "duplicate source rows. The diagnostic is not valid if "
        "no duplicates appear."
    )

    # ---------------------------------------------------------
    # Cross-validation split (3 folds)
    # ---------------------------------------------------------
    from sklearn.model_selection import StratifiedKFold

    cv = StratifiedKFold(
        n_splits=3,
        shuffle=True,
        random_state=42,
    )

    leaked_folds = 0

    for fold_id, (train_idx, val_idx) in enumerate(
        cv.split(boot_original_ids, y[bootstrap_indices])
    ):

        train_source_ids = set(
            boot_original_ids[train_idx].tolist()
        )

        val_source_ids = set(
            boot_original_ids[val_idx].tolist()
        )

        overlap = (
            train_source_ids
            & val_source_ids
        )

        print(
            f"\n  Fold {fold_id}:"
        )

        print(
            f"    train size: {len(train_idx)} "
            f"(unique source: {len(train_source_ids)})"
        )

        print(
            f"    val size:   {len(val_idx)} "
            f"(unique source: {len(val_source_ids)})"
        )

        print(
            f"    source row overlap: {len(overlap)}"
        )

        if overlap:
            leaked_folds += 1

    # ---------------------------------------------------------
    # Diagnosis
    # ---------------------------------------------------------
    if leaked_folds > 0:

        print(
            "\n  RESULT: DUPLICATE-SOURCE LEAKAGE DETECTED"
        )

        print(
            f"  {leaked_folds} of 3 CV folds contain rows "
            "whose original source row appears in both "
            "training and validation sets."
        )

        print(
            "  This confirms that bootstrap-then-CV duplicates "
            "inflate CV scores and produce pathological "
            "bootstrap bias."
        )

    else:

        print(
            "\n  RESULT: No duplicate-source leakage detected "
            "in this bootstrap iteration."
        )

        print(
            "  Note: leakage is stochastic. Repeated runs may "
            "reveal it depending on bootstrap draws."
        )

    # ---------------------------------------------------------
    # The diagnostic returns True regardless of outcome.
    # Its purpose is to expose leakage, not to assert
    # a specific statistical property.
    # ---------------------------------------------------------
    return True

def test_performance_smoke_patch():
    """
    Performance smoke test for the parallel bootstrap patch.

    Uses a very small configuration so it completes in seconds:

        n_bootstrap=10
        n_permutations=10
        n_jobs=2
        n_estimators=10
        cv_folds=3

    This is the appropriate configuration for verifying that the
    parallel patch works correctly before running production-scale
    jobs on real BTC data.
    """

    print("\n" + "=" * 60)
    print("TEST: Performance Smoke Test (Parallel Bootstrap)")
    print("=" * 60)

    df = create_test_data(n=200)

    X = df[[
        "feature1",
        "feature2",
        "feature3",
        "feature4",
    ]]

    y = df["future_breakout"].values

    engine = MonteCarloValidationEngine(
        n_bootstrap=10,
        n_permutations=10,
        n_jobs=2,
        n_estimators=10,
        cv_folds=3,
        random_state=42,
    )

    # Ensure diagnostics are OFF in production path
    engine.bootstrap_diagnostics = False

    import time

    start = time.perf_counter()

    result = engine.compute(X, y)

    elapsed = time.perf_counter() - start

    print(
        f"\n  Smoke test completed in {elapsed:.2f}s"
    )

    print(
        f"  Observed score: {result.observed_score:.4f}"
    )

    print(
        f"  Bootstrap mean: {result.monte_carlo_mean:.4f}"
    )

    print(
        f"  Bootstrap bias: "
        f"{result.robustness_statistics['bootstrap_bias']:.4f}"
    )

    print(
        f"  P-value: {result.p_value:.4f}"
    )

    # ---------------------------------------------------------
    # Structural assertions
    # ---------------------------------------------------------
    assert result.observed_score > 0
    assert 0 <= result.p_value <= 1
    assert not result.bootstrap_distribution.empty
    assert not result.permutation_distribution.empty
    assert isinstance(result.passed, bool)

    print(
        "  ✓ Performance smoke test passed"
    )

    return True

def test_bootstrap_estimator_alignment():
    """
    Bootstrap must use the same estimator as the observed statistic.
    """
    print("\n" + "=" * 60)
    print("TEST: Bootstrap Estimator Alignment")
    print("=" * 60)

    df = create_test_data()

    X = df[['feature1', 'feature2', 'feature3', 'feature4']]
    y = df['future_breakout'].values

    engine = MonteCarloValidationEngine(
        n_bootstrap=30,
        n_permutations=30,
        n_jobs=2,
        fast_mode=True,
        random_state=42,
    )

    observed = engine.compute_score(X, y)
    bootstrap = engine.bootstrap_validation(X, y)

    bias = abs(bootstrap["bias"])

    assert np.isfinite(bias)
    assert abs(bias) < 0.40, (
        f"Bootstrap bias too large: {bias:.4f}. "
        "Observed and bootstrap estimators may not be aligned."
    )

    print(f"  Observed score: {observed:.4f}")
    print(f"  Bootstrap mean: {bootstrap['mean']:.4f}")
    print(f"  Bootstrap bias: {bootstrap['bias']:.4f}")
    print("  ✓ Bootstrap estimator alignment passed")
    return True

def test_true_walk_forward():
    """True walk-forward uses strict temporal separation."""
    print("\n" + "=" * 60)
    print("TEST: True Walk-Forward Validation")
    print("=" * 60)

    df = create_test_data(n=500)

    X = df[['feature1', 'feature2', 'feature3', 'feature4']]
    y = df['future_breakout'].values

    engine = MonteCarloValidationEngine(
        fast_mode=True,
        metric='roc_auc',
    )

    result = engine.walk_forward_stability(
        X,
        y,
        window_size=0.50,
        test_size=0.10,
    )

    assert result["num_windows"] >= 2

    for window in result["windows"]:
        assert window["train_end"] <= window["test_start"]
        assert window["train_end"] == window["test_start"]
        assert (
            window["train_start"]
            < window["train_end"]
            < window["test_end"]
        )

    print(f"  Windows: {result['num_windows']}")
    print(f"  Mean OOS score: {result['mean']:.4f}")
    print("  ✓ True walk-forward test passed")
    return True
