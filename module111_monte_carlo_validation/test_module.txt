def create_test_data(n=500, seed=42):
    """Create synthetic test data with a known weak edge."""
    np.random.seed(seed)

    # Features with a real signal
    x1 = np.random.randn(n)
    x2 = np.random.randn(n) * 0.5 + x1 * 0.3
    x3 = np.random.randn(n) * 0.3 + x1 * 0.2
    x4 = np.random.randn(n)  # Pure noise

    # Target with signal + noise
    y_prob = 1 / (1 + np.exp(-(0.5 * x1 + 0.3 * x2 + 0.1 * x3)))
    y = (np.random.random(n) < y_prob).astype(int)

    # Create DataFrame
    df = pd.DataFrame({
        'feature1': x1,
        'feature2': x2,
        'feature3': x3,
        'feature4': x4,
        'future_breakout': y
    })

    return df

    """Test 1: Compute score."""
    print("\n" + "=" * 60)
    print("TEST 1: Compute Score")
    print("=" * 60)

    y = df['future_breakout'].values

    return True

def test_engine_edge_cases():
    """Test 10: Edge cases."""
    print("\n" + "=" * 60)
    print("TEST 10: Edge Cases")
    print("=" * 60)

    # Test with constant features (should handle gracefully)
    n = 100
    X_constant = pd.DataFrame({
        'f1': np.ones(n),
        'f2': np.ones(n),
        'f3': np.ones(n)
    })
    y_constant = np.random.randint(0, 2, n)

    engine = MonteCarloValidationEngine(fast_mode=True)

    try:
        result = engine.compute(X_constant, y_constant)
        assert result.observed_score >= 0
        print(f"  Constant features handled: score={result.observed_score:.4f}")
    except Exception as e:
        assert False, f"Constant features caused error: {e}"

    # Test with empty data - should raise ValueError
    X_empty = pd.DataFrame()
    y_empty = np.array([])

    try:
        engine.compute_score(X_empty, y_empty)
        assert False, "Empty data should have raised ValueError"
    except ValueError as e:
        assert "Empty dataset" in str(e) or "X and y are required" in str(e)
        print("  ✓ Empty data correctly rejected with ValueError")

    # Test with single sample - should raise appropriate error
    X_single = pd.DataFrame({'f1': [1.0]})
    y_single = np.array([0])

    try:
        engine.compute_score(X_single, y_single)
        assert False, "Single sample should have raised ValueError"
    except ValueError as e:
        assert "at least two classes" in str(e) or "Insufficient samples" in str(e)
        print("  ✓ Single sample correctly rejected")

    print("  ✓ Edge cases test passed")
    return True


def test_bootstrap_estimator_alignment():
    """
    Critical regression test.

    Bootstrap must use the same estimator as the
    observed statistic.

    The previous implementation compared:
        CV ROC-AUC
    against:
        OOB ROC-AUC

    which produced pathological bootstrap bias.
    """

    print("\n" + "=" * 60)
    print("TEST: Bootstrap Estimator Alignment")
    print("=" * 60)

    df = create_test_data()

    X = df[
        [
            'feature1',
            'feature2',
            'feature3',
            'feature4',
        ]
    ]

    y = df[
        'future_breakout'
    ].values

    engine = MonteCarloValidationEngine(
        n_bootstrap=30,
        n_permutations=30,
        n_jobs=2,
        fast_mode=True,
        random_state=42,
    )

    observed = engine.compute_score(
        X,
        y,
    )

    bootstrap = (
        engine.bootstrap_validation(
            X,
            y,
        )
    )

    bias = abs(
        bootstrap["bias"]
    )

    # The exact bias is data dependent.
    # The important regression condition is
    # that it is not the huge estimator mismatch
    # seen in the previous implementation.
    assert np.isfinite(bias)
    assert abs(bias) < 0.40, (
        f"Bootstrap bias too large: {bias:.4f}. "
        "Observed and bootstrap estimators may "
        "not be aligned."
    )

    print(
        f"  Observed score: {observed:.4f}"
    )

    print(
        f"  Bootstrap mean: "
        f"{bootstrap['mean']:.4f}"
    )

    print(
        f"  Bootstrap bias: "
        f"{bootstrap['bias']:.4f}"
    )

    print(
        "  ✓ Bootstrap estimator alignment passed"
    )

    return True

def test_true_walk_forward():

    print("\n" + "=" * 60)
    print("TEST: True Walk-Forward Validation")
    print("=" * 60)

    df = create_test_data(
        n=500
    )

    X = df[
        [
            'feature1',
            'feature2',
            'feature3',
            'feature4',
        ]
    ]

    y = df[
        'future_breakout'
    ].values

    engine = MonteCarloValidationEngine(
        fast_mode=True,
        metric='roc_auc',
    )

    result = engine.walk_forward_stability(
        X,
        y,
        window_size=0.50,
        test_size=0.10,
    )

    assert result[
        "num_windows"
    ] >= 2

    for window in result["windows"]:

        assert (
            window["train_end"]
            <= window["test_start"]
        )

        assert (
            window["train_end"]
            == window["test_start"]
        )

        assert (
            window["train_start"]
            < window["train_end"]
            < window["test_end"]
        )

    print(
        f"  Windows: "
        f"{result['num_windows']}"
    )

    print(
        f"  Mean OOS score: "
        f"{result['mean']:.4f}"
    )

    print(
        "  ✓ True walk-forward test passed"
    )

    return True
